/* Apply modal */
function openApplyModal(e) {
  if (e) e.preventDefault();
  document.getElementById('apply-modal').classList.add('open');
  document.body.style.overflow = 'hidden';
}
function closeApplyModal() {
  document.getElementById('apply-modal').classList.remove('open');
  document.body.style.overflow = '';
}
document.addEventListener('keydown', function(e) {
  if (e.key === 'Escape') closeApplyModal();
});

/* RFB toggle */
function toggleRfb(btn) {
  const item = btn.closest('.rfb-item');
  item.classList.toggle('open');
}
/* Open the RFB referenced in the URL hash, otherwise leave all collapsed */
document.addEventListener('DOMContentLoaded', () => {
  if (location.hash && location.hash.startsWith('#rfb-')) {
    const target = document.querySelector(location.hash);
    if (target) target.classList.add('open');
  }
});

/* Open the target RFB when an in-page RFB link is clicked */
document.addEventListener('click', e => {
  const a = e.target.closest('a.rfb-link');
  if (!a) return;
  const href = a.getAttribute('href');
  if (!href || !href.startsWith('#rfb-')) return;
  const target = document.querySelector(href);
  if (target) target.classList.add('open');
});
